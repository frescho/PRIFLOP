# scripts/pico/CMakeLists.txt
#
# Phase 0 smoke-test build for the sd2iec_pico2w port.
#
# This is deliberately NOT the sd2iec core yet -- it exists only to prove
# the toolchain/Pico SDK/board wiring works end to end (CMake config,
# ninja build, cyw43 init, UART stdio) before any sd2iec source is
# involved.
#
# Once Phase 1+ lands, this CMakeLists.txt will grow into the real build:
# it will add the shared sd2iec core sources (../../src/*.c, the same
# list currently enumerated by scripts/Makefile.main's SRC variable) plus
# src/pico/*.c, and consume the CONFIG_* feature flags via the autoconf.h
# generated by scripts/configparser.pl from configs/config-pico2w (for why CMake owns compilation while
# configparser.pl remains the single source of CONFIG_* truth).
#
# Usage (no PICO_SDK_PATH/toolchain on PATH assumed):
#   cmake -S scripts/pico -B obj-pico2w/cmake-build -G Ninja \
#         -DPICO_SDK_PATH=<path-to-pico-sdk> \
#         -DPICO_BOARD=pico2_w \
#         -DCMAKE_MAKE_PROGRAM=<path-to-ninja> \
#         -DCMAKE_C_COMPILER=<path-to-arm-none-eabi-gcc> \
#         -DCMAKE_CXX_COMPILER=<path-to-arm-none-eabi-g++>
#   cmake --build obj-pico2w/cmake-build

cmake_minimum_required(VERSION 3.13)

if(NOT DEFINED PICO_BOARD)
  set(PICO_BOARD pico2_w CACHE STRING "Board type" FORCE)
endif()

# picotool.exe: the official prebuilt Windows release (v2.3.0-0) was
# flagged as a trojan by Windows Defender, 2026-09-01 -- a known, Raspberry-Pi-acknowledged false positive tied
# to how the official Windows build is compiled (MinGW/MSYS2 binaries
# linking the older MSVCRT runtime trip Defender's heuristics), not an
# actual compromise. Rather than trust another prebuilt binary, picotool
# was instead built locally from the public github.com/raspberrypi/
# picotool source (tag 2.3.0, statically linked against a locally-built
# libusb so USB commands like `load`/`reboot` work) using a freshly
# installed MinGW-w64 UCRT toolchain, then rescanned with Windows
# Defender (clean) before being placed at the path CMake expects below
# -- for the full build recipe. Note:
# `reboot -f`/`load -f` (auto-force a running device into BOOTSEL) is
# currently unreliable against this port's own firmware -- picotool
# reads back a garbled USB serial-number string and never sees the
# device reappear, so it silently fails without resetting anything.
# Harmless (falls back to asking for a manual BOOTSEL press) but not
# yet root-caused; a real fix belongs in this port's own USB descriptor
# setup, not here.
include(pico_sdk_import.cmake)

project(sd2iec_pico2w_phase0 C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

# Workaround: arm-none-eabi-gcc 15.2.1 (the toolchain shipped with this
# Pico SDK 2.3.0 install) defaults -Wimplicit-function-declaration to a
# hard error in C99+ modes (a GCC 14+ behaviour change). pico-sdk 2.3.0's
# own src/common/pico_util/datetime.c is missing "#include <time.h>" and
# relies on the implicit declarations of localtime_r/mktime, so it fails
# to build with this toolchain as soon as anything pulls pico_util in
# (e.g. depending on hardware_timer/pico_time, which sd2iec's arch-timer.c
# needs for time_us_32()/busy_wait_us_32()). This is an upstream SDK/
# toolchain version mismatch, not sd2iec_pico2w code; demote it back to a
# warning here rather than patching the shared ~/.pico-sdk checkout.
# Discovered + verified during (arch-timer.c bring-up).
add_compile_options(-Wno-error=implicit-function-declaration -Wno-error=int-conversion)

# Flash size. Empty: the board's own (pico2_w: 4MB). -DPRIFLOP_FLASH_MB=16 builds
# for a 16MB board: the linker region and PICO_FLASH_SIZE_BYTES in C both follow,
# and src/pico/flash_layout.h sizes the volume from it. Untested on hardware
# (R7c); a board flashed with one size loses its volume to another.
set(PRIFLOP_FLASH_MB "" CACHE STRING "Flash size in MB (empty: the board's)")
if(PRIFLOP_FLASH_MB)
  if(NOT PRIFLOP_FLASH_MB MATCHES "^(2|4|8|16)$")
    message(FATAL_ERROR "PRIFLOP_FLASH_MB must be 2, 4, 8 or 16, not ${PRIFLOP_FLASH_MB}")
  endif()
  math(EXPR PICO_FLASH_SIZE_BYTES "${PRIFLOP_FLASH_MB} * 1024 * 1024")
  add_compile_definitions(PICO_FLASH_SIZE_BYTES=${PICO_FLASH_SIZE_BYTES})
endif()

  # The SDK default is 2048 bytes per core. littlefs nests several frames
  # deep in lfs_dir_commit()/lfs_dir_compact(), writing into a mounted D64
  # adds d64ops on top, and lfs_file_sync() on a 174KB image underneath --
  # and a reset with no watchdog armed means a fault. Measured via
  # /api/stats (system.c's stack painting) rather than guessed at.
  # Core 0's stack lives in SCRATCH_Y, which is 4KB, and core 1's in
  # SCRATCH_X, also 4KB (memmap_default.ld). 4096 is therefore the most
  # core 0 can have without a custom linker script: the SDK's script will
  # happily place a larger .stack_dummy in SCRATCH_Y WITHOUT erroring,
  # leaving __StackBottom pointing below the region at memory that
  # belongs to core 1 -- verified in the .map file, and it makes any
  # stack measurement meaningless as well as unsafe.
  #
  # 2048 (the SDK default) was measurably too little for this firmware;
  # see /api/stats and system.c's stack painting.
  #
  # Core 1's stack was 2048 too, and raised to 4096 on 2026-09-20 -- all of
  # SCRATCH_X, so the same ceiling applies (the .map shows .stack1 filling
  # 0x20080000-0x20081000 exactly). Reason: lwIP's mDNS responder, which runs
  # on core 1 in station mode, has 1100- and 2060-byte stack frames
  # (mdns_handle_response, mdns_handle_probe_tiebreaking) and the plain call
  # chain down to the first is 1768 bytes before its callees. One consequence
  # to know about: SCRATCH_X's unused upper half used to sit between core 1's
  # stack and core 0's (SCRATCH_Y), quietly giving core 0 a 2 KB cushion if it
  # ever overflowed; that gap is gone, so core 0's own high-water mark
  # (/api/stats stack_used of stack_total) is worth watching.
  target_compile_definitions(priflop PRIVATE
    PICO_STACK_SIZE=4096
    PICO_CORE1_STACK_SIZE=4096
  )

  target_compile_options(priflop PRIVATE
    -iquote${AUTOCONF_DIR}
    -iquote${CMAKE_CURRENT_SOURCE_DIR}/../../src
    -iquote${CMAKE_CURRENT_SOURCE_DIR}/../../src/pico
  -iquote${CMAKE_CURRENT_SOURCE_DIR}/../../src/littlefs
    # Normally supplied by scripts/Makefile.main's CDEFS (derived from
    # MAJOR/MINOR/PATCHLEVEL in that file) -- hardcoded here since this
    # throwaway check bypasses the real Make-based version generation.
    -DVERSION=\"0.0-phase1test\"
    -DLONGVERSION=\"-pico2w-phase1test\"
  )

  # (see pico_stdlib's own CMakeLists.txt -- pico_enable_stdio_uart(0)
  # does NOT remove this dependency, it only affects which backend
  # stdio_init_all() wires up). hardware_uart's compiled uart.c defines
  # a real "uart_init" symbol that collides with sd2iec's own
  # src/pico/uart.c at the linker level (same name, different
  # signature, C has no overloading) -- see uart_hw_shim.h/.c for the
  # full story and why uart_hw_shim.c now programs UART registers
  # directly instead of calling the SDK's uart_init(). Linking the
  # specific pieces pico_stdlib would have provided, minus hardware_uart
  # and pico_stdio (sd2iec doesn't use stdio_init_all()/printf either).
  # iec_rx.pio -> iec_rx.pio.h, consumed by src/pico/iec_pio.c
  pico_generate_pio_header(priflop
    ${CMAKE_CURRENT_SOURCE_DIR}/../../src/pico/iec_rx.pio)
  pico_generate_pio_header(priflop
    ${CMAKE_CURRENT_SOURCE_DIR}/../../src/pico/iec_tx.pio)
  pico_generate_pio_header(priflop
    ${CMAKE_CURRENT_SOURCE_DIR}/../../src/pico/primo_turbo.pio)

    pico_time
    pico_util
    pico_platform

    hardware_sync

    hardware_sync_spin_lock
    hardware_xip_cache
    # multicore_launch_core1() itself (src/pico/core1.c, backlog item 4)
    # -- pico_flash was also linked here for a time (ata.c/arch-eeprom.c
    # briefly used its flash_safe_execute()) but that was reverted
    # 2026-09-03 after it broke real C64 IEC timing; see ata.c's
    # commit_slot() comment.
    pico_multicore

    # lwIP-backed variant for the web file-manager -- "poll" mode (not threadsafe_background) so networking is
    # driven cooperatively from src/iec.c's mainloop via network_poll(),
    # matching this whole system's single-core/no-locking design (see
    # src/pico/httpd.c's file header). Needs lwipopts.h reachable on the
    # include path -- already covered by the -iquote src/pico below.
    pico_cyw43_arch_lwip_poll
    pico_lwip_mdns
    # USB CDC debug console (uart_hw_shim.c) -- no debug probe or
    # USB-serial adapter available for Phase 2 testing (2026-08-22), so
    # the same USB cable used to flash the board doubles as a serial
    # console. Doesn't depend on hardware_uart (see uart_hw_shim.c), so
    # this is unrelated to the --allow-multiple-definition situation
    # below.
    pico_stdio
    pico_stdio_usb

  pico_enable_stdio_uart(priflop 0)
  pico_enable_stdio_usb(priflop 1)
  # Without this, boot-time prints (the "sd2iec VERSION #addr" banner
  # etc.) get silently dropped if written before a terminal has actually
  # opened the CDC port -- tud_cdc_connected() is false and
  # stdio_usb_out_chars() just discards them rather than buffering, and
  # Pico boots fast enough that "wait for the COM port to enumerate,
  # then open a terminal" reliably loses the race otherwise (confirmed:
  # first capture attempt during Phase 2 testing, 2026-08-22, came back
  # empty). -1 (block in stdio_init_all() until a host actually opens the
  # port) was used originally, but real-hardware testing on 2026-08-22
  # found this makes the board *completely IEC-dead on every power-up*
  # unless a terminal happens to already be attached with DTR asserted --
  # the IEC mainloop never starts because main() never gets past
  # stdio_init_all(), so the C64 sees nothing on the bus at all ("DEVICE
  # NOT PRESENT"). This is fatal for a device meant to work standalone,
  # so it's now a bounded wait: enough to usually catch a boot banner if
  # a terminal is already open, but boot (and the IEC bus) always
  # proceeds regardless.
  # littlefs: all three buffers are supplied statically by lfs_flash.c, so
  # it must never reach for the heap. LFS_NO_MALLOC turns an accidental
  # allocation into a compile error rather than a runtime failure deep
  # inside a filesystem call. LFS_NO_DEBUG/WARN drop its logging, which
  # would otherwise pull in printf and write to a console this firmware
  # uses for IEC-critical timing.
  target_compile_definitions(priflop PRIVATE
    LFS_NO_MALLOC
    LFS_NO_DEBUG
    LFS_NO_WARN
  )

  target_compile_definitions(priflop PRIVATE
    # Was 2000, and it cost two seconds of EVERY boot.
    #
    # stdio_init_all() blocks here waiting for a USB host to assert DTR.
    # On a drive that lives plugged into a C64 rather than a laptop, that
    # host never arrives, so the full timeout was paid every time --
    # measured 2026-09-06 as 0.0s to 2.0s of a 2.0s pre-bus init, i.e.
    # the ENTIRE delay before the drive would answer the C64. Everything
    # else in main() -- spi, timer, i2c, bus_init, rtc, disk, config,
    # filesystem -- came to under 0.1s combined.
    #
    # The wait existed so early output would not be lost to a console
    # that attaches a moment later. bootlog.c makes that moot: every
    # character is recorded in RAM that survives a reset and can be read
    # back over HTTP or dumped when a console does attach. Waiting for a
    # reader is no longer how the early log is kept.
    PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0
    # A stalled USB console must not be able to hang the IEC bus.
    #
    # The SDK default is 500000 -- half a second of blocking PER CHARACTER
    # when the host is not draining the endpoint. uart_putc() is called
    # from the bus mainloop, so at the default a console that stops
    # reading turns any logging into a guaranteed watchdog reset: eight
    # characters is already past the 8s watchdog. This is the same hazard
    # the console_lock comment in uart.c describes, left at its dangerous
    # default until it actually bit (2026-09-06).
    #
    # 2ms keeps output working normally for an attached, draining console
    # -- writes complete in microseconds when anyone is listening -- while
    # bounding the damage from one that is not.
    PICO_STDIO_USB_STDOUT_TIMEOUT_US=2000
  )
  # Optional compile-time fallback access-point name/password for
  # src/pico/netconfig.c (the web file-manager) --
  # the device always creates its own Wi-Fi AP (see network.c for why
  # AP mode rather than joining an existing network). /PRIFLOP.CFG on
  # the flash FAT volume (netconfig.c) is now the primary way to set
  # this at runtime; these CMake variables are only the fallback used
  # when that file is absent or doesn't set a given key, so a
  # from-scratch build with no flash content yet still has a way to
  # preset something other than the hardcoded "PRIFLOP"/open default
  # -- pass with:
  #   cmake ... -DWIFI_SSID=YourAPName -DWIFI_PASSWORD=YourPassword
  # (password must be 8+ characters or the AP falls back to open, since
  # WPA2 requires it -- see network.c). Omit both and it just uses the
  # defaults. NOTE: these are cached CMake variables -- once set, they
  # stick across reconfigures until explicitly unset with
  # `cmake -UWIFI_SSID -UWIFI_PASSWORD`. A stale cached value from an
  # earlier test build silently overrode this default for a long time
  # in one past session -- if the AP's SSID
  # ever doesn't match what's expected here, check CMakeCache.txt for
  # this before suspecting the code.
  if(DEFINED WIFI_SSID AND DEFINED WIFI_PASSWORD)
    target_compile_definitions(priflop PRIVATE
      WIFI_SSID=\"${WIFI_SSID}\"
      WIFI_PASSWORD=\"${WIFI_PASSWORD}\"
    )
  endif()
  # Optional compile-time fallback HTTP Basic Auth password for the web
  # UI (2026-09-02, see httpd.c/network.c/netconfig.c) -- same pattern
  # as WIFI_SSID/WIFI_PASSWORD above: /PRIFLOP.CFG's `ADMIN=` key is now
  # the primary way to set this, this CMake variable is only the
  # fallback. Same cached-variable caveat applies (`cmake
  # -UADMIN_PASSWORD` to clear it). Pass with:
  #   cmake ... -DADMIN_PASSWORD=YourPassword
  # Omit it (and leave `/PRIFLOP.CFG` without an `ADMIN=` line) and the
  # web UI stays open (this server's original, pre-auth behavior).
  if(DEFINED ADMIN_PASSWORD)
    target_compile_definitions(priflop PRIVATE
      ADMIN_PASSWORD=\"${ADMIN_PASSWORD}\"
    )
  endif()
  # hardware_uart's compiled uart.c still ends up in the link regardless
  # of the library selection above: pico_runtime itself unconditionally
  # links hardware_uart (see its CMakeLists.txt), and pico_runtime's
  # crt0/vector-table/linker-script wiring is not something to route
  # around -- unlike pico_stdlib, it's not optional. Since nothing in
  # this build calls the SDK's real uart_init by name any more (see
  # uart_hw_shim.c), and CMake places this target's own sources before
  # transitively-linked library sources in the link command (confirmed
  # by the linker error naming src/pico/uart.c's uart_init as the
  # "first defined" one), --allow-multiple-definition keeps our
  # intentional uart_init and discards the SDK's unused one -- the
  # standard, well-precedented way embedded projects override an SDK-
  # provided symbol under the same name (e.g. newlib syscall overrides),
  # not a hack specific to this collision.
  target_link_options(priflop PRIVATE -Wl,--allow-multiple-definition)
  # The UF2's "absolute block" (the SDK's workaround for RP2350-A2 erratum
  # E10, added whenever PICO_RP2350_A2_SUPPORTED) defaults to 0x10ffff00: the
  # last page of a 16MB address space. This chip has 4MB, the address wraps,
  # and the bootloader erased the last sector -- FLASH_EEPROM_OFFSET, where
  # XW keeps extension hiding, the device address and the rest -- on every
  # drag-and-drop flash. "A reflash resets the settings" was that, from
  # 2026-09-07 until it was measured on 2026-10-02: a setting survived two
  # flashes of a UF2 without the block and was wiped by two with it
  # Moved, not dropped, so an A2 board keeps the workaround: the
  # last page of the 1MB firmware region (FLASH_UF2_ABS_BLOCK, flash_layout.h),
  # which the ~490KB image is nowhere near. The SDK appends its own
  # --abs-block only when the target does not already pass one. On a 16MB
  # board the SDK's default would land on the settings sector itself.
  set_target_properties(priflop PROPERTIES
    PICOTOOL_EXTRA_UF2_ARGS "--abs-block;0x100fff00")
  pico_add_extra_outputs(priflop)
endif()
